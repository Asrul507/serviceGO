async function handleBookingSubmit(e) { e.preventDefault(); const cartKeys = Object.keys(cart); if (cartKeys.length === 0) { alert('Keranjang Anda masih kosong. Silakan pilih minimal 1 layanan.'); return; } const btn = document.getElementById('submitBtn'); btn.disabled = true; btn.innerHTML = ' Memproses Pesanan...'; const itemsPayload = cartKeys.map(k => ({ id: Number(cart[k].id), name: cart[k].name, price: Number(cart[k].price), qty: Number(cart[k].qty) })); const paymentMethod = document.querySelector('input[name="payment_option"]:checked').value; const payload = { customer_name: document.getElementById('custName').value, customer_phone: document.getElementById('custPhone').value, customer_address: document.getElementById('custAddress').value, items: itemsPayload, scheduled_date: document.getElementById('orderDate').value, payment_method: paymentMethod, notes: document.getElementById('orderNotes').value }; try { const res = await fetch('/api/orders', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); const data = await res.json(); if (res.ok && data.success) { showPaymentInstruction(data.order_id, data.total_price, paymentMethod); } else { alert('Gagal: ' + (data.error || 'Terjadi kesalahan saat memproses pesanan')); btn.disabled = false; btn.innerHTML = ' Pesan Sekarang'; } } catch (err) { alert('Terjadi kendala jaringan.'); btn.disabled = false; btn.innerHTML = ' Pesan Sekarang'; } }
